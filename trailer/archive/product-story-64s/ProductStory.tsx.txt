import React from 'react';
import {AbsoluteFill, CanvasImage, Easing, interpolate, OffthreadVideo, Series, staticFile, useCurrentFrame, useVideoConfig} from 'remotion';
import {ThePile, BaskyWide} from './LandscapeEdit';

import {clamp, Phone} from './style';

// Source cues are real recorded UI events from isolated, labelled demo profiles.
// Later-load montage does not imply one 25-item batch unlocks every mountain.
export const STORY_CUES:Record<string,number>=recordedCues;
const ease={...clamp,easing:Easing.bezier(.18,.8,.18,1)};
const base:React.CSSProperties={fontFamily:'Nunito',fontWeight:900,color:'white',background:'#071b42',overflow:'hidden'};
function Backdrop({mountain='ben',dark=true}: {mountain?:string;dark?:boolean}) {
 const f=useCurrentFrame();const {fps}=useVideoConfig();
 const src=mountain==='ben'?'art/landing-highland-wide.webp':`art/playable-${mountain}-landscape.webp`;
 return <AbsoluteFill><CanvasImage src={staticFile(src)} premountFor={fps} style={{width:'100%',height:'100%',objectFit:'cover',transform:`scale(${interpolate(f,[0,180],[1.02,1.10],clamp)})`,transformOrigin:'65% 55%'}}/><AbsoluteFill style={{background:dark?'linear-gradient(90deg,#061a32e8 0%,#061a32ba 48%,#061a3233 78%)':'linear-gradient(90deg,#f4fbfff5,#f4fbffb0 52%,#f4fbff00 77%)'}}/></AbsoluteFill>;
}
function StepCopy({title,detail,eyebrow}: {title:React.ReactNode;detail:string;eyebrow?:string}) {
 const f=useCurrentFrame();
 return <div style={{position:'absolute',left:100,top:250,width:1050,opacity:interpolate(f,[0,9],[0,1],clamp),transform:`translateY(${interpolate(f,[0,17],[32,0],ease)}px)`}}>
  {eyebrow&&<div style={{fontSize:29,color:'#ffc26c',marginBottom:27,letterSpacing:2}}>{eyebrow}</div>}
  <div style={{fontSize:132,letterSpacing:-4,lineHeight:1.025,textShadow:'0 4px 24px #04172866'}}>{title}</div>
  <div style={{fontSize:40,lineHeight:1.3,marginTop:38,maxWidth:965,color:'#e9f7ff'}}>{detail}</div>
 </div>;
}
function Punch({children}: {children:React.ReactNode}) {
 return <span style={{fontFamily:'Summit Block',fontWeight:900,fontSize:134,letterSpacing:-4,color:'#b8f0d9',textTransform:'uppercase'}}>{children}</span>;
}
function PortraitFootage({from=0,rate=1,x=1220}: {from?:number;rate?:number;x?:number}) {
 const f=useCurrentFrame();const {fps}=useVideoConfig();
 return <div style={{transform:`translateY(${interpolate(f,[0,15],[55,0],ease)}px)`}}><Phone x={x} y={68} width={448} aspectRatio={.5}><OffthreadVideo src={staticFile('recordings/story-portrait.webm')} trimBefore={Math.round(from*fps)} playbackRate={rate} muted style={{width:'100%',height:'100%'}}/></Phone></div>;
}

  <CanvasImage src={staticFile('art/reference-logo.webp')} premountFor={fps} style={{position:'absolute',left:280,top:105,width:620,height:620,objectFit:'contain',transform:`scale(${interpolate(f,[0,20],[.9,1],ease)})`}}/>
  <div style={{position:'absolute',left:75,top:730,width:1070,textAlign:'center',fontSize:80,lineHeight:1.06,color:'#071b42',letterSpacing:-3}}>Conquer your pile of laundry.<br/><span style={{color:'#00765d'}}>One load at a time.</span></div>
  <div style={{transform:`translateX(${interpolate(f,[0,22],[220,0],ease)}px)`}}><Phone src="captures/ben-home.jpg" x={1240} y={85} width={418}/></div>
